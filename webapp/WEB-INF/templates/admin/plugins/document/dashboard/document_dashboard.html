<#-- A document as a row of a flush list: its publication as a dot, its title, its type, state and date on one line, the preview and the correction as icons (when the workflow offers them), every action of the workflow in the menu --> <#macro documentDashboardRow document> <#local documentStatus><#if document.publishedStatus=0>#i18n{document.dashboard.PublishedStatus}<#else>#i18n{document.dashboard.UnPublishedStatus}
${documentStatus}
${document.id} · ${document.title}
${document.type} · ${document.state} · #i18n{document.dashboard.labelDateModification} ${document.dateModification}
<#if document.comment?has_content>
${document.comment}
<#list document.actions?filter( action -> action.url?exists && ( action.iconUrl! == 'eye' || action.iconUrl! == 'pencil' ) ) as action> <@icon style=action.iconUrl />${action.name!}
#i18n{document.dashboard.columnActions}
<#list document.actions as action> <#if action.url?exists> <@icon style=action.iconUrl! class='me-2' /> ${action.name!}
<#if last_modified_document??> <@adminDashboardWidget title='#i18n{document.dashboard.columnLastModifiedDocument}' id='site_dashboard_document_last_modified' url=url boxIcon='file-pencil' iconColor='orange' flush=true> <@documentDashboardRow document=last_modified_document /> <#if last_published_document??> <@adminDashboardWidget title='#i18n{document.dashboard.columnLastPublishedDocument}' id='site_dashboard_document_last_published' url=url boxIcon='world' iconColor='green' flush=true> <@documentDashboardRow document=last_published_document />
