<#ftl output_format="HTML"> <#-- # Macros to display a document in the templates of the document types. # This file uses the HTML output format: the values are HTML-escaped whatever the auto-escaping setting of the site. # # Model of a document template: # document the document (DocumentView): id, codeDocumentType, title, summary, dateValidityBegin, dateValidityEnd, # categories (list of names), validated, attributes (map of the values by attribute code) # document_type the document type, with its attributes (code, name, codeAttributeType) # front office only: publication_date (may be null), site_locale, request_parameters (map) # # An attribute value is a string, or a file (url, fileName, contentType, image) for the file and image attributes. # An attribute without value is absent from document.attributes: test it with ?? before using it. --> <#-- documentValue # Print a plain text value, HTML-escaped # param value the value, may be null --> <#macro documentValue value=''>${value!} <#-- documentHtml #
